-- ============================================
-- PASO 2: CREAR TABLAS
-- ============================================

-- TABLA PACIENTE
CREATE TABLE PACIENTE (
    ID_Paciente NUMBER PRIMARY KEY,
    Nombre VARCHAR2(50),
    Apellido VARCHAR2(50),
    DNI VARCHAR2(10) UNIQUE,
    Telefono VARCHAR2(15),
    Email VARCHAR2(80),
    Direccion VARCHAR2(40)
);

-- TABLA TIPO PAGO
CREATE TABLE TIPO_PAGO (
    ID_Tipo_Pago NUMBER PRIMARY KEY,
    Descripcion VARCHAR2(50),
    Monto NUMBER(6,2)
);

-- TABLA ESPECIALIDAD
CREATE TABLE ESPECIALIDAD (
    ID_Especialidad NUMBER PRIMARY KEY,
    Nombre_Especialidad VARCHAR2(50),
    Descripcion VARCHAR2(30),
    Dias_Atencion VARCHAR2(30)
);

-- TABLA MEDICO
CREATE TABLE MEDICO (
    ID_Medico NUMBER PRIMARY KEY,
    Nombre VARCHAR2(50),
    Apellido VARCHAR2(50),
    DNI VARCHAR2(10) UNIQUE,
    Telefono VARCHAR2(9),
    Email VARCHAR2(50),
    Horario_Atencion VARCHAR2(20),
    ID_Especialidad NUMBER,
    CONSTRAINT fk_medico_especialidad FOREIGN KEY (ID_Especialidad)
        REFERENCES ESPECIALIDAD(ID_Especialidad)
);

-- TABLA PERSONAL CALL CENTER
CREATE TABLE PERSONAL_CALL_CENTER (
    ID_Personal NUMBER PRIMARY KEY,
    Nombre VARCHAR2(50),
    Apellido VARCHAR2(50),
    DNI NUMBER(10),
    Turno VARCHAR2(20)
);

-- TABLA CITA
CREATE TABLE CITA (
    ID_Cita NUMBER PRIMARY KEY,
    Fecha DATE,
    Hora_Inicio VARCHAR2(10),
    Hora_Fin VARCHAR2(10),
    Estado VARCHAR2(20), -- pendiente, atendida, cancelada
    Monto_Cobrado NUMBER(6,2),
    ID_Paciente NUMBER,
    ID_Medico NUMBER,
    ID_Tipo_Pago NUMBER,
    ID_Personal NUMBER,
    CONSTRAINT fk_cita_paciente FOREIGN KEY (ID_Paciente)
        REFERENCES PACIENTE(ID_Paciente),
    CONSTRAINT fk_cita_medico FOREIGN KEY (ID_Medico)
        REFERENCES MEDICO(ID_Medico),
    CONSTRAINT fk_cita_tipopago FOREIGN KEY (ID_Tipo_Pago)
        REFERENCES TIPO_PAGO(ID_Tipo_Pago),
    CONSTRAINT fk_cita_personal FOREIGN KEY (ID_Personal)
        REFERENCES PERSONAL_CALL_CENTER(ID_Personal)
);

-- TABLA PAGO MEDICO
CREATE TABLE PAGO_MEDICO (
    ID_Pago NUMBER PRIMARY KEY,
    Mes VARCHAR2(15),
    Año NUMBER(4),
    Total_Citas NUMBER,
    Monto_Total NUMBER(10,2),
    Fecha_Pago DATE,
    Estado VARCHAR2(20),
    ID_Medico NUMBER,
    CONSTRAINT fk_pago_medico FOREIGN KEY (ID_Medico)
        REFERENCES MEDICO(ID_Medico)
);

-- SECUENCIAS PARA CLAVES PRIMARIAS
CREATE SEQUENCE seq_paciente START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE seq_tipo_pago START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE seq_especialidad START WITH 1 INCREMENT BY 1;

CREATE SEQUENCE seq_cita START WITH 1 INCREMENT BY 1;
